ECONOMY OVERVIEW — KPI CARDS
-- Total money in economy (wallet + bank)
SELECT
  SUM(JSON_UNQUOTE(JSON_EXTRACT(accounts, '$.money'))) + 
  SUM(JSON_UNQUOTE(JSON_EXTRACT(accounts, '$.bank')))
  AS total_economy
FROM players;

-- Money earned today (from transaction logs)
SELECT SUM(amount) AS earned_today
FROM economy_transactions
WHERE type = 'earn'
  AND created_at >= CURDATE();

-- Money spent today
SELECT SUM(amount) AS spent_today
FROM economy_transactions
WHERE type = 'spend'
  AND created_at >= CURDATE();
7-DAY FLOW CHART
-- Daily earn vs spend for last 7 days
SELECT
  DATE(created_at) AS day,
  SUM(CASE WHEN type = 'earn'
       THEN amount ELSE 0 END) AS earned,
  SUM(CASE WHEN type = 'spend'
       THEN amount ELSE 0 END) AS spent
FROM economy_transactions
WHERE created_at >= DATE_SUB(NOW(), INTERVAL 7 DAY)
GROUP BY DATE(created_at)
ORDER BY day ASC;

TOP EARNING JOBS
-- Total daily earnings grouped by job
SELECT
  p.job                   AS job_name,
  SUM(t.amount)           AS total_earned,
  COUNT(DISTINCT t.identifier) AS player_count
FROM economy_transactions t
JOIN players p ON p.identifier = t.identifier
WHERE t.type = 'earn'
  AND t.created_at >= CURDATE()
GROUP BY p.job
ORDER BY total_earned DESC
LIMIT 10;

-- Column 'job' may be 'job_name' in your
-- players table. Check your ESX/QB schema.
RICHEST PLAYERS
-- Top 10 players by total wealth
SELECT
  p.name,
  p.identifier,
  p.job,
  (
    JSON_UNQUOTE(JSON_EXTRACT(p.accounts, '$.money')) +
    JSON_UNQUOTE(JSON_EXTRACT(p.accounts, '$.bank'))
  ) AS total_wealth,
  ROUND(p.playtime / 60) AS hours_played
FROM players p
ORDER BY total_wealth DESC
LIMIT 10;

MONEY SINKS
-- Where money leaves economy today
SELECT
  category,
  SUM(amount) AS total_spent
FROM economy_transactions
WHERE type = 'spend'
  AND created_at >= CURDATE()
GROUP BY category
ORDER BY total_spent DESC;

ECONOMY HEALTH METRICS
-- Average player wallet
SELECT AVG(
  JSON_UNQUOTE(JSON_EXTRACT(accounts, '$.money')) +
  JSON_UNQUOTE(JSON_EXTRACT(accounts, '$.bank'))
) AS avg_wealth
FROM players;

-- Illegal vs legal earn ratio
SELECT
  SUM(CASE WHEN source_type = 'illegal'
       THEN amount ELSE 0 END) * 100.0
  / SUM(amount) AS illegal_pct
FROM economy_transactions
WHERE type = 'earn'
  AND created_at >= CURDATE();
